No OCR-supplied URLs or styles. */ window.renderUserOcr = (target, pages, text) => { const element = (tag, value) => { const out = document.createElement(tag); if (value !== undefined) out.textContent = value; return out; }; function inline(target, value) { const pattern = /(\*\*([^*]+)\*\*|`([^`]+)`)/g; let at = 0, match; while ((match = pattern.exec(value))) { target.append(document.createTextNode(value.slice(at, match.index))); target.append(element(match[2] ? null : element('pre'); if (pre) target.append(pre); list = null; continue; } if (pre) { pre.append(document.createTextNode(line + '\n')); continue; } if (line.includes('|') && divider(lines[i + 1] || '')) { const table = element('table'), header = element('tr'); for (const cell of cells(line)) { const th = element('th'); inline(th, cell); header.append(th); } const head = element('thead'); head.append(header); table.append(head); i++; const body = element('tbody'); while (i + 1 < lines.length && lines[i + 1].includes('|') && lines[i + 1].trim()) { const row = element('tr'); for (const cell of cells(lines[++i])) { const td = element('td'); inline(td, cell); row.append(td); } body.append(row); } table.append(body); const scroll = element('div'); scroll.className = 'table-scroll'; scroll.append(table); target.append(scroll); list = null; continue; } const heading = /^(#{1,4})\s+(.+)$/.exec(line), bullet = /^\s*(?:[-*+]\s+|\d+[.)]\s+)(.+)$/.exec(line); if (heading) { const h = element('h' + heading[1].length); inline(h, heading[2]); target.append(h); list = null; } else if (bullet) { if (!list) { list = element('ul'); target.append(list); } const li = element('li'); inline(li, bullet[1]); list.append(li); } else { list = null; if (line.trim()) { const p = element('p'); inline(p, line); target.append(p); } } } } target.replaceChildren(); const content = pages?.length ? pages : text ?
